// Auflösung zu Türchen 24: Die Weiße-Weihnachten-Station
// Verdrahtung siehe tag-24.md
// Bibliotheken: Adafruit BMP280 Library, DHT sensor library, Adafruit Unified Sensor

#include <Wire.h>
#include <Adafruit_BMP280.h>
#include <DHT.h>

const char ART[] PROGMEM = R"ART(
      *        *         *
               *
       *      /.\     *
             /..'\
        *    /'.'\       *
            /.''.'\
     *      /.'.'.\    *
           /'.''.'.\
           ^^^[_]^^^
  F R O H E   W E I H N A C H T E N
  Tuerchen 24: Die Weisse-Weihnachten-Station
)ART";

Adafruit_BMP280 bmp;        // KY-052: Temperatur und Luftdruck, I2C-Adresse 0x77
const int DHT_PIN = 2;      // KY-015: hier nur für die Luftfeuchte
DHT dht(DHT_PIN, DHT11);

const int LED_ROT = 9;      // KY-016 von Tag 5
const int LED_GRUEN = 6;
const int LED_BLAU = 5;

void setzeFarbe(int r, int g, int b) {
  analogWrite(LED_ROT, r);
  analogWrite(LED_GRUEN, g);
  analogWrite(LED_BLAU, b);
}

// Augenzwinkernde Schneechance: kalt, feucht und tiefer Luftdruck klingen nach Schnee
int schneechance(float t, float feuchte, float druck) {
  int chance = 0;
  if (t <= 2.0) chance += 40;
  if (t <= 0.0) chance += 20;
  if (feuchte >= 80.0) chance += 25;
  if (druck < 1005.0) chance += 15;
  return min(chance, 100);
}

void setup() {
  pinMode(LED_ROT, OUTPUT);
  pinMode(LED_GRUEN, OUTPUT);
  pinMode(LED_BLAU, OUTPUT);
  Serial.begin(9600);
  Serial.println((const __FlashStringHelper*)ART);
  dht.begin();
  if (!bmp.begin(0x77)) {
    Serial.println("BMP280 nicht gefunden - Verdrahtung prüfen.");
  }
}

void loop() {
  float t = bmp.readTemperature();           // °C
  float druck = bmp.readPressure() / 100.0;  // Pa -> hPa
  float feuchte = dht.readHumidity();        // %
  if (isnan(feuchte)) {
    Serial.println("DHT11: Lesefehler, neuer Versuch ...");
    delay(2000);
    return;
  }

  int chance = schneechance(t, feuchte, druck);
  if (chance >= 60) {
    setzeFarbe(255, 255, 255);  // schneeweiß
  } else if (chance >= 30) {
    setzeFarbe(0, 120, 255);    // hellblau: vielleicht
  } else {
    setzeFarbe(0, 255, 0);      // grüne Weihnachten
  }

  Serial.print("Temperatur: ");
  Serial.print(t, 1);
  Serial.print(" C | Luftfeuchte: ");
  Serial.print(feuchte, 0);
  Serial.print(" % | Luftdruck: ");
  Serial.print(druck, 1);
  Serial.print(" hPa | Schneechance: ");
  Serial.print(chance);
  Serial.println(" %");
  delay(2000);  // der DHT11 liefert höchstens alle 2 Sekunden einen neuen Wert
}
